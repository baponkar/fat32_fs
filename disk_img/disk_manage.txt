# List all loop devices
losetup -a

# Or filter for your specific image
losetup -a | grep disk_1.img

# Check mount points
mount | grep loop

# Find next available loop device
sudo losetup -f  # Shows next free loop (e.g., /dev/loop5)

# Manually set up loop with offset
sudo losetup -o $((2048*512)) /dev/loop5 disk_img/disk_1.img

# Now mount it
sudo mount /dev/loop5 /mnt1

-----------------------------------------------------------------
# First, unmount if already mounted
sudo umount /mnt1 2>/dev/null

# Detach all loop devices for this image
sudo losetup -d /dev/loop0   # or whatever loop# was used

# Or detach ALL loop devices (careful!)
sudo losetup -D

# Now try again
sudo mount -o loop,offset=$((2048*512)) disk_img/disk_1.img /mnt1
